    const[cur,ytdP,d30P]=await Promise.all([fetchLatest(t),fetchClose(t,YTD_START),fetchClose(t,d30ago)]);
    if(!cur||!ytdP||!d30P) continue;
    ytdNum+=((cur-ytdP)/ytdP*100)*w;
    d30Num+=((cur-d30P)/d30P*100)*w;
    totalW+=w;
  }
  return totalW>0?{ytd:ytdNum/totalW,d30:d30Num/totalW}:null;
}

async function refreshAll(){
  const d30=get30DAgo();
  for(const pid of['def','agg','hc']){
    const layers=PORTFOLIOS[pid];
    let portYTD=0,port30=0;
    for(const[lname,ldata]of Object.entries(layers)){
      const sfx=lname.replace(/ /g,'_').toLowerCase();
      const res=await calcLayer(ldata.tickers,d30);
      if(res){
        const el30=document.getElementById(pid+'_30d_'+sfx);
        const elYTD=document.getElementById(pid+'_ytd_'+sfx);
        if(el30) el30.innerHTML=fmtPct(res.d30);
        if(elYTD) elYTD.innerHTML=fmtPct(res.ytd);
        portYTD+=res.ytd*(ldata.alloc/100);
        port30+=res.d30*(ldata.alloc/100);
      }
    }
    const totYTD=document.getElementById(pid+'_ytd_total');
    const tot30=document.getElementById(pid+'_30d_total');
    if(totYTD) totYTD.innerHTML=fmtTotal(portYTD);
    if(tot30)  tot30.innerHTML=fmtTotal(port30);
    const ts=document.getElementById(pid+'_ts');
    if(ts){
      const open=await isMarketOpen();
      const status=open?' · Market Open':' · Market Closed';
      ts.textContent='Updated '+new Date().toLocaleTimeString()+status;
    }
    for(const[btick,bkey]of[['SPY','spy'],['QQQ','qqq']]){
      const[bcur,bYTD,b30d]=await Promise.all([fetchLatest(btick),fetchClose(btick,YTD_START),fetchClose(btick,d30)]);
      if(bcur&&bYTD){const bel=document.getElementById(pid+'_ytd_'+bkey);if(bel)bel.innerHTML=fmtPct((bcur-bYTD)/bYTD*100);}
      if(bcur&&b30d){const bel=document.getElementById(pid+'_30d_'+bkey);if(bel)bel.innerHTML=fmtPct((bcur-b30d)/b30d*100);}
    }
  }
}

refreshAll();
setInterval(()=>{Object.keys(priceCache).filter(k=>k.includes('_latest')).forEach(k=>delete priceCache[k]);refreshAll();},900000);
})();
